# Flappy CP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Flappy Bird clone in the Cigma Points game hub that costs 5 CP to play, credits +1 CP per edible eaten live to the user's balance, and caps at 3 runs per calendar day (America/New_York reset).

**Architecture:** Client renders on `<canvas>` inside the existing `GameDialog`. A pure `engine.ts` runs physics; a shared `flappy-seed.ts` module produces the pipe/edible layout from a server-issued seed so the tRPC `flappyRouter` can validate each eat as claim-once against a known layout. Placeholder-rectangle sprites in v1; a `sprites.ts` indirection layer swaps to real PNGs later with a single-file change.

**Tech Stack:** Next.js 16, React 19, tRPC v11.17, Prisma v7.8 (Postgres/Neon), Zod v4, Vitest, Playwright, Canvas 2D. Package manager: `npm`.

- **Commit style:** short imperative subject, no scope prefix (matches recent commits like `same modal everywhere`, `intelligent hide on bottom bar`). Do not include a Co-Authored-By trailer unless the user asks.

```ts
// Deterministic PRNG — same seed → same sequence, on both client and server.
// This is what makes claim-once anti-cheat cheap: server can reproduce the exact
// edible layout the client sees.

/** mulberry32 — 32-bit state, well-diffused, tiny. See
 *  https://github.com/bryc/code/blob/master/jshash/PRNGs.md#mulberry32 */
export function mulberry32(seed: number): () => number {
  let a = seed >>> 0
  return () => {
    a = (a + 0x6d2b79f5) >>> 0
    let t = a
    t = Math.imul(t ^ (t >>> 15), t | 1)
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61)
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
  }
}
```

```ts
// Shared server + client layout. Given a seed, produces the same infinite sequence
// of pipes and edibles on both sides — which is what makes claim-once anti-cheat
// possible in the flappy router. Deliberately imports only pure modules so it can
// run in Node (server) without pulling any DOM code.

import { mulberry32 } from "@/components/game-hub/flappy/rng"
import { GAP_HEIGHT, PIPE_INTERVAL_PX, PIPE_WIDTH_PX } from "@/components/game-hub/flappy/constants"

export type Pipe = { index: number; xCenter: number; gapY: number }
export type Edible = { index: number; x: number; y: number }
export type CanvasSize = { width: number; height: number }

const START_OFFSET_PX = 300 // first pipe appears this far right of the bird
const GAP_VERTICAL_MARGIN = 60 // gap center must be at least this far from top/bottom

/** Deterministic gap y-center for a given (seed, index) tied to the canvas height. */
function gapYFor(seed: number, index: number, height: number): number {
  // Reseed per-index so we get well-distributed y positions without keeping state.
  const rand = mulberry32(seed ^ (index * 2654435761))
  const min = GAP_VERTICAL_MARGIN + GAP_HEIGHT / 2
  const max = height - GAP_VERTICAL_MARGIN - GAP_HEIGHT / 2
  return min + rand() * (max - min)
}

export function getPipe(seed: number, index: number, canvas: CanvasSize): Pipe {
  return {
    index,
    xCenter: START_OFFSET_PX + index * PIPE_INTERVAL_PX,
    gapY: gapYFor(seed, index, canvas.height),
  }
}

export function getEdible(seed: number, index: number, canvas: CanvasSize): Edible {
  const p = getPipe(seed, index, canvas)
  return { index, x: p.xCenter, y: p.gapY }
}

export function sliceForWindow(
  seed: number,
  scrollX: number,
  canvas: CanvasSize,
): { pipes: Pipe[]; edibles: Edible[] } {
  const minX = scrollX - PIPE_WIDTH_PX
  const maxX = scrollX + canvas.width + PIPE_WIDTH_PX
  // Invert: index = (x - START_OFFSET_PX) / PIPE_INTERVAL_PX
  const first = Math.max(0, Math.floor((minX - START_OFFSET_PX) / PIPE_INTERVAL_PX))
  const last = Math.ceil((maxX - START_OFFSET_PX) / PIPE_INTERVAL_PX)
  const pipes: Pipe[] = []
  const edibles: Edible[] = []
  for (let i = first; i <= last; i++) {
    const p = getPipe(seed, i, canvas)
    if (p.xCenter >= minX && p.xCenter <= maxX) {
      pipes.push(p)
      edibles.push({ index: i, x: p.xCenter, y: p.gapY })
    }
  }
  return { pipes, edibles }
}
```

```ts
// Pure physics + collision. No React, no DOM, no tRPC — so this file is fully
// unit-testable and the same rules apply on every device.

import {
  BIRD_SIZE_PX,
  BIRD_X_FRAC,
  FLAP_VELOCITY,
  GAP_HEIGHT,
  GRAVITY,
  MOUTH_OPEN_MS,
  PIPE_WIDTH_PX,
  SCROLL_SPEED,
} from "./constants"
import { sliceForWindow, type CanvasSize } from "@/lib/flappy-seed"

export type World = {
  seed: number
  canvas: CanvasSize
  bird: { x: number; y: number; vy: number; tilt: number; mouthOpenUntil: number }
  scrollX: number
  eatenLocal: Set<number>
  status: "ready" | "playing" | "over"
  startedAt: number
}

export function createWorld(seed: number, canvas: CanvasSize, now: number): World {
  return {
    seed,
    canvas,
    bird: {
      x: canvas.width * BIRD_X_FRAC,
      y: canvas.height / 2,
      vy: 0,
      tilt: 0,
      mouthOpenUntil: 0,
    },
    scrollX: 0,
    eatenLocal: new Set<number>(),
    status: "ready",
    startedAt: now,
  }
}

export function flap(world: World, now: number): void {
  if (world.status === "over") return
  if (world.status === "ready") {
    world.status = "playing"
    world.startedAt = now
  }
  world.bird.vy = FLAP_VELOCITY
  world.bird.mouthOpenUntil = now + MOUTH_OPEN_MS
}

function birdRect(w: World): { x: number; y: number; size: number } {
  return { x: w.bird.x - BIRD_SIZE_PX / 2, y: w.bird.y - BIRD_SIZE_PX / 2, size: BIRD_SIZE_PX }
}

function rectsOverlap(
  ax: number, ay: number, aw: number, ah: number,
  bx: number, by: number, bw: number, bh: number,
): boolean {
  return ax < bx + bw && ax + aw > bx && ay < by + bh && ay + aw > by + 0 && ay + ah > by
}

export function step(world: World, dtSec: number, now: number): { eatenNow: number[] } {
  const eatenNow: number[] = []
  if (world.status !== "playing") return { eatenNow }

  // Physics
  world.bird.vy += GRAVITY * dtSec
  world.bird.y += world.bird.vy * dtSec
  world.bird.tilt = Math.atan2(world.bird.vy, 300)
  world.scrollX += SCROLL_SPEED * dtSec

  // Floor / ceiling
  if (world.bird.y >= world.canvas.height || world.bird.y <= 0) {
    world.status = "over"
    return { eatenNow }
  }

  // Pipe + edible collisions (bird is at fixed screen x; pipe x is in world coords,
  // so we compare against pipe.xCenter - scrollX to get its on-screen x).
  const { pipes, edibles } = sliceForWindow(world.seed, world.scrollX, world.canvas)
  const b = birdRect(world)

  for (const p of pipes) {
    const pipeScreenX = p.xCenter - world.scrollX - PIPE_WIDTH_PX / 2
    // Top pipe: y=[0, gapY - GAP_HEIGHT/2). Bottom pipe: y=[gapY + GAP_HEIGHT/2, canvas.height].
    const topH = p.gapY - GAP_HEIGHT / 2
    const bottomY = p.gapY + GAP_HEIGHT / 2
    const bottomH = world.canvas.height - bottomY
    if (rectsOverlap(b.x, b.y, b.size, b.size, pipeScreenX, 0, PIPE_WIDTH_PX, topH)) {
      world.status = "over"
      return { eatenNow }
    }
    if (rectsOverlap(b.x, b.y, b.size, b.size, pipeScreenX, bottomY, PIPE_WIDTH_PX, bottomH)) {
      world.status = "over"
      return { eatenNow }
    }
  }

  for (const e of edibles) {
    if (world.eatenLocal.has(e.index)) continue
    const eScreenX = e.x - world.scrollX
    // Bird-vs-edible: distance between centers.
    const dx = world.bird.x - eScreenX
    const dy = world.bird.y - e.y
    if (Math.hypot(dx, dy) < BIRD_SIZE_PX / 2 + 12) {
      world.eatenLocal.add(e.index)
      world.bird.mouthOpenUntil = Math.max(world.bird.mouthOpenUntil, now + MOUTH_OPEN_MS)
      eatenNow.push(e.index)
    }
  }

  return { eatenNow }
}
```

```ts
// Flappy CP router — server-authoritative CP debits/credits + anti-cheat validation
// against a deterministic seeded layout. Mirrors the wordleRouter pattern for auth
// and notifyCpChange integration.
//
// Anti-cheat model: on start, we issue a seed and a runId. The client renders the
// layout that flappy-seed.ts produces from that seed; when the player eats edible
// N, they call eat({ runId, edibleIndex: N }) and the server verifies N against
// the same layout, that N hasn't been claimed already, and that the eat rate is
// physically plausible. See docs/superpowers/specs/2026-07-17-flappy-cp-design.md
// for the full contract.

import { TRPCError } from "@trpc/server"
import { randomInt } from "crypto"
import { after } from "next/server"
import { createTRPCRouter, protectedProcedure } from "@/trpc/init"
import { db } from "@/lib/db"
import { dayKey } from "@/lib/day-key"
import { notifyCpChange } from "@/lib/notifications"
import {
  ENTRY_COST,
  PLAYS_PER_DAY,
  SWEEP_ACTIVE_AFTER_MS,
} from "@/components/game-hub/flappy/constants"

export const flappyRouter = createTRPCRouter({
  /** Drives the hub card: runs left today + whether the user can play right now. */
  getStatus: protectedProcedure.query(async ({ ctx }) => {
    const userId = ctx.session.user.id
    const day = dayKey()
    const [runsToday, user] = await Promise.all([
      db.flappyRun.count({ where: { userId, day } }),
      db.user.findUnique({ where: { id: userId }, select: { cigmaPoints: true } }),
    ])
    const runsRemaining = Math.max(0, PLAYS_PER_DAY - runsToday)
    const canPlay = runsRemaining > 0 && (user?.cigmaPoints ?? 0) >= ENTRY_COST
    return { day, runsToday, runsRemaining, canPlay }
  }),

  /** Debits 5 CP and issues a run token + seed. Firing this is the "you're
   *  committed" moment — the client fires it on first flap, not on modal open,
   *  so opening the dialog costs nothing. */
  start: protectedProcedure.mutation(async ({ ctx }) => {
    const userId = ctx.session.user.id
    const day = dayKey()

    const created = await db.$transaction(async (tx) => {
      // 1. Sweep this user's stale ACTIVE rows (no refund — they had their shot).
      await tx.flappyRun.updateMany({
        where: {
          userId,
          status: "ACTIVE",
          startedAt: { lt: new Date(Date.now() - SWEEP_ACTIVE_AFTER_MS) },
        },
        data: { status: "ABANDONED", endedAt: new Date() },
      })

      // 2. Enforce the daily cap.
      const runsToday = await tx.flappyRun.count({ where: { userId, day } })
      if (runsToday >= PLAYS_PER_DAY) {
        throw new TRPCError({ code: "FORBIDDEN", message: "Daily play limit reached." })
      }

      // 3. Enforce balance ≥ ENTRY_COST.
      const user = await tx.user.findUnique({ where: { id: userId }, select: { cigmaPoints: true } })
      if ((user?.cigmaPoints ?? 0) < ENTRY_COST) {
        throw new TRPCError({ code: "FORBIDDEN", message: "Not enough CP to play." })
      }

      // 4. Debit balance.
      await tx.user.update({
        where: { id: userId },
        data: { cigmaPoints: { decrement: ENTRY_COST } },
      })

      // 5. Insert the run with a fresh seed.
      const seed = randomInt(0, 2 ** 32)
      const row = await tx.flappyRun.create({
        data: { userId, day, seed, entryCost: ENTRY_COST, status: "ACTIVE" },
        select: { id: true, seed: true },
      })
      return { row, runsToday: runsToday + 1 }
    })

    after(() => notifyCpChange(userId))

    return {
      runId: created.row.id,
      seed: created.row.seed,
      entryCost: ENTRY_COST,
      runsRemaining: Math.max(0, PLAYS_PER_DAY - created.runsToday),
    }
  }),
})
```

```ts
  /** Claim one eaten edible. Hot path — must be cheap. Anti-cheat is the point. */
  eat: protectedProcedure
    .input(z.object({ runId: z.string(), edibleIndex: z.number().int().min(0) }))
    .mutation(async ({ ctx, input }) => {
      const userId = ctx.session.user.id
      // Canvas size is only needed to derive the edible position for validation.
      // Any width/height with the same aspect ratio yields identical (index, x-y ratio)
      // relationships — the anti-cheat cares about existence + rate, not pixels.
      const CANVAS = { width: 360, height: 360 / CANVAS_ASPECT_RATIO }

      const result = await db.$transaction(async (tx) => {
        const run = await tx.flappyRun.findUnique({
          where: { id: input.runId },
          select: { id: true, userId: true, seed: true, status: true, startedAt: true, eatsClaimed: true, cpEarned: true },
        })
        if (!run || run.userId !== userId) {
          throw new TRPCError({ code: "FORBIDDEN", message: "Run not found." })
        }
        if (run.status !== "ACTIVE") {
          throw new TRPCError({ code: "FORBIDDEN", message: "Run is not active." })
        }

        // Validate index against the deterministic layout for this seed. The layout
        // is infinite (index → deterministic pipe/edible), so `getEdible` succeeds
        // for any non-negative int — the Zod schema `.min(0)` is the real existence
        // check. We still call it so future non-uniform layouts (e.g. a "no edible
        // in this gap" seed) would throw here without a router change.
        getEdible(run.seed, input.edibleIndex, CANVAS)

        if (run.eatsClaimed.includes(input.edibleIndex)) {
          throw new TRPCError({ code: "CONFLICT", message: "Edible already claimed." })
        }

        // Rate limit: physical minimum time between eats.
        const elapsedMs = Date.now() - run.startedAt.getTime()
        const claimedCount = run.eatsClaimed.length
        if (claimedCount > 0 && elapsedMs / claimedCount < MIN_MS_PER_EDIBLE) {
          throw new TRPCError({ code: "TOO_MANY_REQUESTS", message: "Eating too fast." })
        }

        const updated = await tx.flappyRun.update({
          where: { id: run.id },
          data: {
            eatsClaimed: { push: input.edibleIndex },
            cpEarned: { increment: 1 },
          },
          select: { cpEarned: true },
        })
        await tx.user.update({
          where: { id: userId },
          data: { cigmaPoints: { increment: 1 } },
        })
        return { cpEarned: updated.cpEarned }
      })

      // Deliberately no notifyCpChange — would spam email/push. `end` sends one summary.
      return result
    }),
```

```ts
// The only file that knows about canvas drawing. v1: colored primitives. v2: swap
// the branches inside `draw` to `ctx.drawImage(...)` calls against images loaded in
// `preload`. Nothing else in the tree changes.

import type { World } from "./engine"
import {
  BIRD_SIZE_PX,
  EDIBLE_RADIUS_PX,
  GAP_HEIGHT,
  PIPE_WIDTH_PX,
} from "./constants"
import { sliceForWindow } from "@/lib/flappy-seed"

const SKY = "#38bdf8"      // Tailwind sky-400
const PIPE = "#22c55e"     // Tailwind green-500
const PIPE_DARK = "#15803d" // Tailwind green-700 (rim)
const BIRD = "#facc15"     // Tailwind yellow-400
const BIRD_OUTLINE = "#a16207" // Tailwind yellow-700
const EDIBLE = "#ffffff"

/** v1 no-op. In v2 this becomes `Promise.all([loadImage(...), ...])`. */
export async function preload(): Promise<void> {
  return
}

export function draw(ctx: CanvasRenderingContext2D, world: World, now: number): void {
  const { canvas, bird, scrollX, seed } = world

  // Background
  ctx.fillStyle = SKY
  ctx.fillRect(0, 0, canvas.width, canvas.height)

  // Pipes + edibles
  const { pipes, edibles } = sliceForWindow(seed, scrollX, canvas)
  for (const p of pipes) {
    const x = p.xCenter - scrollX - PIPE_WIDTH_PX / 2
    const topH = p.gapY - GAP_HEIGHT / 2
    const bottomY = p.gapY + GAP_HEIGHT / 2
    const bottomH = canvas.height - bottomY

    ctx.fillStyle = PIPE
    ctx.fillRect(x, 0, PIPE_WIDTH_PX, topH)
    ctx.fillRect(x, bottomY, PIPE_WIDTH_PX, bottomH)

    // Rim on the exposed edges of the pipes for a bit of depth.
    ctx.fillStyle = PIPE_DARK
    ctx.fillRect(x - 2, topH - 8, PIPE_WIDTH_PX + 4, 8)
    ctx.fillRect(x - 2, bottomY, PIPE_WIDTH_PX + 4, 8)
  }

  for (const e of edibles) {
    if (world.eatenLocal.has(e.index)) continue
    const ex = e.x - scrollX
    ctx.fillStyle = EDIBLE
    ctx.beginPath()
    ctx.arc(ex, e.y, EDIBLE_RADIUS_PX, 0, Math.PI * 2)
    ctx.fill()
  }

  // Bird — tilted rounded square with a small "mouth" indicator that opens on flap/eat.
  ctx.save()
  ctx.translate(bird.x, bird.y)
  ctx.rotate(bird.tilt)
  ctx.fillStyle = BIRD
  ctx.fillRect(-BIRD_SIZE_PX / 2, -BIRD_SIZE_PX / 2, BIRD_SIZE_PX, BIRD_SIZE_PX)
  ctx.strokeStyle = BIRD_OUTLINE
  ctx.lineWidth = 2
  ctx.strokeRect(-BIRD_SIZE_PX / 2, -BIRD_SIZE_PX / 2, BIRD_SIZE_PX, BIRD_SIZE_PX)
  // Mouth-open cue: a small dark rectangle on the leading edge while mouthOpenUntil > now.
  if (bird.mouthOpenUntil > now) {
    ctx.fillStyle = "#0f172a"
    ctx.fillRect(BIRD_SIZE_PX / 4, -4, 8, 8)
  }
  ctx.restore()
}
```

```tsx
"use client"

// FlappyGame — the <canvas> host. Owns the requestAnimationFrame loop and the
// input listeners; forwards eat/end events up via callbacks so the parent can
// wire tRPC. Refs everywhere: React state would re-render the whole tree at
// 60fps, which is exactly what we're avoiding by using canvas.

import { useEffect, useRef } from "react"
import { createWorld, flap, step } from "./engine"
import { draw, preload } from "./sprites"
import { CANVAS_ASPECT_RATIO } from "./constants"

type Props = {
  seed: number
  liveScore: number
  onFirstFlap: () => void
  onEat: (edibleIndex: number) => void
  onEnd: (finalScore: number) => void
}

export function FlappyGame({ seed, liveScore, onFirstFlap, onEat, onEnd }: Props) {
  const canvasRef = useRef<HTMLCanvasElement | null>(null)
  const wrapperRef = useRef<HTMLDivElement | null>(null)
  const flapPendingRef = useRef(false)
  const startedRef = useRef(false)
  const endedRef = useRef(false)
  const liveScoreRef = useRef(liveScore)

  // Keep the ref current so the RAF loop reads the freshest HUD number.
  useEffect(() => { liveScoreRef.current = liveScore }, [liveScore])

  useEffect(() => {
    const canvas = canvasRef.current
    const wrapper = wrapperRef.current
    if (!canvas || !wrapper) return

    // Size the canvas to its wrapper (portrait aspect ratio applied via CSS).
    const dpr = Math.min(window.devicePixelRatio || 1, 2)
    const rect = wrapper.getBoundingClientRect()
    const cssWidth = rect.width
    const cssHeight = rect.width / CANVAS_ASPECT_RATIO
    canvas.width = Math.round(cssWidth * dpr)
    canvas.height = Math.round(cssHeight * dpr)
    canvas.style.width = `${cssWidth}px`
    canvas.style.height = `${cssHeight}px`

    const ctx = canvas.getContext("2d")
    if (!ctx) return
    ctx.scale(dpr, dpr)

    let raf = 0
    let lastNow = performance.now()
    const world = createWorld(seed, { width: cssWidth, height: cssHeight }, lastNow)

    function frame(now: number) {
      const dtSec = Math.min((now - lastNow) / 1000, 0.033) // clamp big gaps (tab hidden)
      lastNow = now

      if (flapPendingRef.current) {
        const wasReady = world.status === "ready"
        flap(world, now)
        if (wasReady && world.status === "playing" && !startedRef.current) {
          startedRef.current = true
          onFirstFlap()
        }
        flapPendingRef.current = false
      }

      const { eatenNow } = step(world, dtSec, now)
      for (const idx of eatenNow) onEat(idx)

      // Draw current live score into the HUD region — done via CSS overlay outside
      // the canvas; drawScene handles the game itself.
      draw(ctx, world, now)

      if (world.status === "over" && !endedRef.current) {
        endedRef.current = true
        onEnd(liveScoreRef.current)
      }

      raf = requestAnimationFrame(frame)
    }
    raf = requestAnimationFrame(frame)

    // Input — pointerdown for immediate mobile response; keydown for desktop.
    const requestFlap = () => { flapPendingRef.current = true }
    const onPointerDown = (e: PointerEvent) => { e.preventDefault(); requestFlap() }
    const onKeyDown = (e: KeyboardEvent) => {
      if (e.metaKey || e.ctrlKey || e.altKey) return
      if (e.key === " " || e.key === "ArrowUp") { e.preventDefault(); requestFlap() }
    }
    canvas.addEventListener("pointerdown", onPointerDown)
    window.addEventListener("keydown", onKeyDown)

    // Kick off async preload but don't gate the loop (v1 preload is a no-op).
    void preload()

    return () => {
      cancelAnimationFrame(raf)
      canvas.removeEventListener("pointerdown", onPointerDown)
      window.removeEventListener("keydown", onKeyDown)
      // Fire end if the user closed the modal mid-run.
      if (startedRef.current && !endedRef.current) {
        endedRef.current = true
        onEnd(liveScoreRef.current)
      }
    }
  }, [seed, onEat, onEnd, onFirstFlap])

  return (
    <div ref={wrapperRef} className="relative mx-auto w-full max-w-[360px]">
      <canvas
        ref={canvasRef}
        style={{ touchAction: "none" }}
        className="block w-full rounded-2xl ring-1 ring-border bg-sky-400 select-none"
        aria-label="Flappy CP game canvas"
      />
      {/* Live-earned CP HUD, top-left */}
      <div className="pointer-events-none absolute left-3 top-3 rounded-md bg-black/40 px-2 py-1 text-sm font-semibold text-white">
        +{liveScore} CP
      </div>
    </div>
  )
}
```

```tsx
"use client"

// Flappy — card + dialog + tRPC wiring. This file owns the modal state, the four
// tRPC calls (getStatus / start / eat / end), and the "runs remaining" card copy.
// Rendering + physics live in <FlappyGame />; anti-cheat lives in the router.

import { useState } from "react"
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query"
import { useSession } from "next-auth/react"
import { Bird } from "lucide-react"
import { useTRPC } from "@/trpc/client"
import { GameCard } from "@/components/game-hub/game-card"
import { GameDialog } from "@/components/game-hub/game-dialog"
import { FlappyGame } from "./flappy-game"
import { ENTRY_COST } from "./constants"

export function Flappy({ index = 0 }: { index?: number }) {
  const [open, setOpen] = useState(false)
  const [runId, setRunId] = useState<string | null>(null)
  const [seed, setSeed] = useState<number | null>(null)
  const [liveScore, setLiveScore] = useState(0)
  const [finalSummary, setFinalSummary] = useState<{ cpEarned: number; net: number } | null>(null)

  const trpc = useTRPC()
  const qc = useQueryClient()
  const { status: authStatus } = useSession()

  const statusQ = useQuery(
    trpc.flappy.getStatus.queryOptions(undefined, {
      enabled: authStatus === "authenticated",
    }),
  )
  const runsRemaining = statusQ.data?.runsRemaining ?? 0
  const canPlay = statusQ.data?.canPlay ?? false

  const start = useMutation(trpc.flappy.start.mutationOptions())
  const eat = useMutation(trpc.flappy.eat.mutationOptions())
  const end = useMutation(trpc.flappy.end.mutationOptions())

  const hint =
    authStatus !== "authenticated"
      ? "Sign in to play"
      : runsRemaining === 0
        ? "Back tomorrow"
        : !canPlay
          ? `Need ${ENTRY_COST} CP`
          : `${runsRemaining} left today · ${ENTRY_COST} CP`

  function openDialog() {
    setFinalSummary(null)
    setLiveScore(0)
    setSeed(null)
    setRunId(null)
    setOpen(true)
  }

  function closeDialog() {
    setOpen(false)
  }

  const [startError, setStartError] = useState<string | null>(null)

  async function handleFirstFlap() {
    setStartError(null)
    try {
      const res = await start.mutateAsync()
      setRunId(res.runId)
      setSeed(res.seed)
      void qc.invalidateQueries(trpc.user.getMe.queryFilter())
    } catch (err) {
      // Spec §7: on start failure, no debit occurred; show a message and let the user close.
      setStartError(err instanceof Error ? err.message : "Could not start run.")
      setOpen(false)
    }
  }

  function handleEat(edibleIndex: number) {
    if (!runId) return
    setLiveScore((s) => s + 1) // optimistic
    eat.mutate({ runId, edibleIndex }, {
      onSuccess: () => {
        void qc.invalidateQueries(trpc.user.getMe.queryFilter())
      },
    })
  }

  function handleEnd() {
    if (!runId) return
    // Spec §7: retry once with a 1s backoff. If both attempts fail, we log — the
    // 5-minute server sweep will still mark the run ABANDONED, and the next
    // user.getMe query will reconcile balance.
    const tryEnd = () =>
      end.mutateAsync({ runId }).then((data) => {
        setFinalSummary({ cpEarned: data.cpEarned, net: data.net })
        void qc.invalidateQueries(trpc.user.getMe.queryFilter())
        void qc.invalidateQueries(trpc.flappy.getStatus.queryFilter())
      })
    tryEnd().catch(() => {
      window.setTimeout(() => {
        tryEnd().catch((err) => {
          console.warn("flappy end mutation failed twice", err)
        })
      }, 1000)
    })
  }

  return (
    <>
      <GameCard
        icon={Bird}
        name="Flappy CP"
        hint={hint}
        available={canPlay && authStatus === "authenticated"}
        index={index}
        onClick={openDialog}
        ariaLabel="Flappy CP — tap to fly, eat to earn"
      />

      <GameDialog
        open={open}
        onOpenChange={setOpen}
        title="Flappy CP"
        description={`Tap to flap. Each edible = +1 CP. Costs ${ENTRY_COST} CP per run.`}
      >
        {/* If we don't yet have a seed, keep the "ready" hero pre-flight seed=0. FlappyGame
            will call onFirstFlap the moment the user taps; parent then swaps in the real seed. */}
        {finalSummary ? (
          <div className="mx-auto flex max-w-sm flex-col items-center gap-3 py-6 text-center">
            <p className="text-lg font-semibold">Run over</p>
            <p className="text-3xl font-bold text-emerald-500">
              +{finalSummary.cpEarned} CP · net {finalSummary.net >= 0 ? `+${finalSummary.net}` : finalSummary.net}
            </p>
            <p className="text-sm text-muted-foreground">
              {statusQ.data?.runsRemaining ?? 0} runs left today
            </p>
            <div className="flex gap-2">
              <button
                type="button"
                onClick={openDialog}
                disabled={!canPlay}
                className="rounded-md bg-primary px-4 py-2 text-sm font-semibold text-primary-foreground disabled:opacity-50"
              >
                Play again
              </button>
              <button
                type="button"
                onClick={closeDialog}
                className="rounded-md bg-muted px-4 py-2 text-sm font-semibold"
              >
                Close
              </button>
            </div>
          </div>
        ) : (
          <FlappyGame
            key={seed ?? 0}
            seed={seed ?? 0}
            liveScore={liveScore}
            onFirstFlap={() => void handleFirstFlap()}
            onEat={handleEat}
            onEnd={handleEnd}
          />
        )}
      </GameDialog>
    </>
  )
}
```
